find_available_moves(position p, board b[0...n-1][0...n-1],&size)
    index=0;
    //right
    if(p.x<n and b[p.x+1][p.y]==-1)
        available_moves[index].x=p.x+1
        available_moves[index].y = p.y
        b[p.x+1][p.y]=1
        index++
    //left
    if(p.x>0 and b[p.x-1][p.y]==-1)
        available_moves[index].x=p.x-1
        available_moves[index].y = p.y
        b[p.x+1][p.y]=1
        index++
    //downward
     if(p.y>0 and b[p.x][p.y-1]==-1)
        available_moves[index].x=p.x
        available_moves[index].y = p.y-1
        b[p.x][p.y-1]=1
        index++
    //upward
     if(p.y<n and b[p.x][p.y+1]==-1)
        available_moves[index].x=p.x
        available_moves[index].y = p.y+1
        b[p.x][p.y+1]=1

   size=index+1
